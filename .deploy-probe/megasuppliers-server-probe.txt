    public static function showcategory()
    {
        $category_model = new shopCategoryModel();
        $categories = $category_model->getFullTree('id, left_key, right_key, parent_id, depth, name, count, type, status, include_sub_categories');
        if (!empty($categories)) {
            foreach ($categories as &$item) {
                if (!isset($item['children_count'])) {
                    $item['children_count'] = 0;
                }
                if (!isset($item['total_count'])) {
                    $item['total_count'] = $item['count'];
                }
                if (isset($categories[$item['parent_id']])) {
                    $parentcat = &$categories[$item['parent_id']];
                    if (!isset($parentcat['children_count'])) {
                        $parentcat['children_count'] = 0;
                    }
                    if (!isset($parentcat['total_count'])) {
                        $parentcat['total_count'] = $parentcat['count'];
                    }
                    ++$parentcat['children_count'];
                    $parentcat['total_count'] += $item['count'];
                    unset($parentcat);
                }
            }
            unset($item);
            return $categories;
        }
    }

            if ($type_id) {
                $sql="SELECT id as product_id FROM shop_product where type_id = '".$type_id."'";
            }

            if ($set_id) {
                $sql="SELECT product_id FROM shop_set_products where set_id = '".$set_id."'";
            }

            if ($set_id && $type_id) {
                $sql="SELECT id as product_id FROM `shop_product` AS t1 LEFT JOIN `shop_set_products` AS t2 ON t1.id = t2.product_id WHERE t1.type_id='".$type_id."' AND t2.set_id='".$set_id."'";
            }
        }

        return $this->response = array(
            'countImages' => (isset($sql)) ? $images->query("SELECT * FROM shop_product_images WHERE product_id IN (" . $sql . ")")->count() : $images->countAll(),
        );
    }

class shopPlugincontrolPluginBackendController extends waJsonController
{
    public function execute()
    {
        if (!$this->getUser()->getRights('shop', 'settings')) {
            throw new waException(_w('Access denied'));
        }
        if (waRequest::post('id')=='run') {
            $this->generateConfig();
        }

    private function generateConfig()
    {
        $plugins = waRequest::post();
        unset($plugins['id']);
        foreach ($plugins as &$value) {
            $value = (bool) $value;
        }
        $app_config = wa()->getConfig()->getAppConfig('shop');
        $path = $app_config->getConfigPath('plugins.php', true);
        $plugin_php = include $path;
        foreach ($plugins as $plugin => $state) {
            $plugin_php[$plugin] = $state;
        }
        waUtils::varExportToFile($plugin_php, $path, true);
        if (wa()->appExists('installer')) {
            wa('installer');
            installerHelper::flushCache();
        }

    protected function checkUpdates()
    {
        $is_from_template = waConfig::get('is_template');
        $disable_exception_log = waConfig::get('disable_exception_log');
        waConfig::set('is_template', null);

        $locking_model = $app_settings_model = new waAppSettingsModel();
        $time = $app_settings_model->get(array($this->app_id, $this->id), 'update_time');

        try {
            if (!$time) {
                $is_first_launch = true;

                // To avoid running install.php multiple times in parallel, obtain a named lock
                // and then read update_time again because it might have changed in another thread
                $locking_model->exec("SELECT GET_LOCK(?, -1)", ["wa_init_plugin_{$this->app_id}_{$this->id}"]);
                $app_settings_model->clearCache(array($this->app_id, $this->id));
                $time = $app_settings_model->get(array($this->app_id, $this->id), 'update_time', null);

                try {
                    if (!$time) {
                        waConfig::set('disable_exception_log', true);
                        $this->install();
                    } else {
                        // All work has already been done in another thread.
                        return;
                    }
                } catch (Exception $e) {
                    waLog::log("Error installing plugin ".$this->app_id.".".$this->id." at first run:\n".$e->getMessage()." (".$e->getCode().")\n".$e->getTraceAsString());
                    throw $e;
                }
            } else {
                $is_first_launch = false;
            }

            $is_debug = waSystemConfig::isDebug();

            if (!$is_debug) {
                $cache = new waVarExportCache('updates', -1, $this->app_id.".".$this->id);
                if ($cache->isCached() && $cache->get() <= $time) {
                    return;
                }
            }
            $path = $this->path.'/lib/updates';
            $cache_database_dir = wa()->getConfig()->getPath('cache').'/db';
            if (file_exists($path)) {
                $iterator = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($path), RecursiveIteratorIterator::SELF_FIRST);
                $files = array();
                foreach ($iterator as $file) {
                    /**
                     * @var SplFileInfo $file
                     */
                    if ($file->isFile() && preg_match('/^[0-9]+\.php$/', $file->getFilename())) {
                        $t = substr($file->getFilename(), 0, -4);
                        if ($t > $time) {
                            $files[$t] = $file->getPathname();
                        }
                    }
                }

                if ($files && !$is_first_launch) {
                    // To avoid running meta-updates in parallel, obtain a named lock
                    // and then read update_time again because it might have changed in another thread
                    $locking_model->exec("SELECT GET_LOCK(?, -1)", ["wa_init_plugin_{$this->app_id}_{$this->id}"]);
                    $app_settings_model->clearCache(array($this->app_id, $this->id));
                    $time = $app_settings_model->get(array($this->app_id, $this->id), 'update_time', null);
                }

                ksort($files);
                if (!$is_debug && !empty($cache)) {
                    // get last time
                    if ($files) {
                        $keys = array_keys($files);
                        $cache->set(end($keys));
                    } else {
                        $cache->set($time ? $time : 1);
                    }
                }
                waConfig::set('disable_exception_log', true);
                foreach ($files as $t => $file) {
                    try {
                        if (!$is_first_launch && $time < $t) {
                            $this->includeUpdate($file);
                            waFiles::delete($cache_database_dir);
                            $app_settings_model->set(array($this->app_id, $this->id), 'update_time', $t);
                        }
                    } catch (Exception $e) {
                        waLog::log("Error running update of plugin {$this->app_id}.{$this->id}: {$file}\n".$e->getMessage()." (".$e->getCode().")\n".$e->getTraceAsString());
                        throw new waException(sprintf(_ws('Error while running update of %s.%s plugin: %s'), $this->app_id, $this->id, $file), 500, $e);
                    }
                }
            } else {
                $t = 1;
            }

            if ($is_first_launch) {
                $app_settings_model->set(array($this->app_id, $this->id), 'update_time', ifempty($t, 1));
            }
        } finally {
            waConfig::set('is_template', $is_from_template);
            waConfig::set('disable_exception_log', $disable_exception_log);
            if (!empty($is_first_launch) || !empty($files)) {
                $locking_model->exec("SELECT RELEASE_LOCK(?)", ["wa_init_plugin_{$this->app_id}_{$this->id}"]);
            }
        }
    }

<?php
class shopMegasuppliersImportService
{
    private $supplier_model;
    private $map_model;
    private $import_model;
    private $sku_model;

    public function __construct()
    {
        $this->supplier_model = new shopMegasuppliersSupplierModel();
        $this->map_model = new shopMegasuppliersProductModel();
        $this->import_model = new shopMegasuppliersImportModel();
        $this->sku_model = new shopProductSkusModel();
    }

    public function importRows($supplier_id, array $rows, array $options = [])
    {
        $supplier_id = (int)$supplier_id;
        $supplier = $this->supplier_model->getById($supplier_id);
        if (!$supplier || empty($supplier['active'])) {
            throw new waException('Поставщик не выбран или отключён.');
        }
        if (!$rows) {
            throw new waException('Нет строк для импорта.');
        }

        $source = isset($options['source']) ? (string)$options['source'] : 'file';
        $filename = isset($options['filename']) ? (string)$options['filename'] : '';
        $create_missing = !empty($options['create_missing']);
        $update_catalog = !array_key_exists('update_catalog', $options) || !empty($options['update_catalog']);
        $field_map = isset($options['field_map']) && is_array($options['field_map']) ? $options['field_map'] : [];

        $now = date('Y-m-d H:i:s');
        $import_id = $this->import_model->insert([
            'supplier_id' => $supplier_id,
            'source' => $source,
            'filename' => $filename,
            'status' => 'running',
            'total' => count($rows),
            'created_count' => 0,
            'updated_count' => 0,
            'linked_count' => 0,
            'skipped_count' => 0,
            'error_count' => 0,
            'errors_json' => '[]',
            'user_id' => wa()->getUser()->getId() ?: 0,
            'created_at' => $now,
            'finished_at' => null,
        ]);

        $stats = ['created' => 0, 'updated' => 0, 'linked' => 0, 'skipped' => 0, 'errors' => []];
        foreach ($rows as $index => $raw) {
            try {
                $item = $this->mapRow($raw, $field_map);
                if ($item['supplier_sku'] === '') {
                    throw new waException('Не указан артикул поставщика.');
                }
                $result = $this->upsertItem($supplier_id, $item, $raw, $create_missing, $update_catalog);
                $stats[$result]++;
            } catch (Exception $e) {
                $stats['errors'][] = ['row' => $index + 2, 'error' => $e->getMessage()];
                if (count($stats['errors']) > 100) {
                    $stats['errors'] = array_slice($stats['errors'], 0, 100);
                }
            }
        }

        $this->import_model->updateById($import_id, [
            'status' => $stats['errors'] ? 'done_with_errors' : 'done',
            'created_count' => $stats['created'],
            'updated_count' => $stats['updated'],
            'linked_count' => $stats['linked'],
            'skipped_count' => $stats['skipped'],
            'error_count' => count($stats['errors']),
            'errors_json' => json_encode($stats['errors'], JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES),
            'finished_at' => date('Y-m-d H:i:s'),
        ]);
        $stats['import_id'] = $import_id;
        $stats['supplier'] = $supplier;
        return $stats;
    }

    private function mapRow(array $row, array $field_map)
    {
        $aliases = [
            'supplier_sku' => ['артикул поставщика', 'артикул', 'sku', 'код', 'код товара', 'vendor code', 'vendor_code', 'article'],
            'external_id' => ['внешний id', 'external id', 'external_id', 'yml id', 'id поставщика'],
            'name' => ['наименование', 'название', 'товар', 'name'],
            'price' => ['цена', 'цена продажи', 'розничная цена', 'price'],
            'purchase_price' => ['закупка', 'закупочная цена', 'себестоимость', 'purchase price', 'purchase_price', 'cost'],
            'compare_price' => ['цена до скидки', 'старая цена', 'compare price', 'compare_price'],
            'stock' => ['остаток', 'наличие', 'количество', 'stock', 'qty'],
        ];
        $item = [];
        foreach ($aliases as $target => $names) {
            $value = '';
            if (!empty($field_map[$target]) && array_key_exists($field_map[$target], $row)) {
                $value = $row[$field_map[$target]];
            } else {
                foreach ($names as $name) {
                    if (array_key_exists($name, $row)) {
                        $value = $row[$name];
                        break;
                    }
                }
            }
            $item[$target] = is_scalar($value) ? trim((string)$value) : '';
        }
        return $item;
    }

    private function upsertItem($supplier_id, array $item, array $raw, $create_missing, $update_catalog)
    {
        $link = $this->map_model->getByField([
            'supplier_id' => $supplier_id,
            'supplier_sku' => $item['supplier_sku'],
        ]);
        $product_id = $link ? (int)$link['product_id'] : 0;
        $sku_id = $link ? (int)$link['sku_id'] : 0;
        $result = $link ? 'updated' : 'linked';

        if (!$product_id || !$sku_id) {
            $matches = $this->sku_model->getByField('sku', $item['supplier_sku'], true);
            if (count($matches) === 1) {
                $sku = reset($matches);
                $product_id = (int)$sku['product_id'];
                $sku_id = (int)$sku['id'];
                $result = 'linked';
            } elseif (count($matches) > 1) {
                throw new waException('Найдено несколько товаров Webasyst с таким артикулом; автоматическая привязка отменена.');
            } elseif ($create_missing) {
                list($product_id, $sku_id) = $this->createProduct($item);
                $result = 'created';
            } else {
                $this->saveLink($supplier_id, 0, 0, $item, $raw);
                return 'skipped';
            }
        }

        if ($update_catalog) {
            $this->updateProduct($product_id, $sku_id, $item);
        }
        $this->saveLink($supplier_id, $product_id, $sku_id, $item, $raw);
        return $result;
    }

    private function createProduct(array $item)
    {
        if ($item['name'] === '') {
            throw new waException('Для создания нового товара требуется название.');
        }
        $type_model = new shopTypeModel();
        $type = $type_model->select('id')->order('sort')->fetchAssoc();
        $currency = wa('shop')->getConfig()->getCurrency();
        $sku_data = [
            'sku' => $item['supplier_sku'],
            'available' => 1,
            'status' => 1,
        ];
        foreach (['price', 'purchase_price', 'compare_price'] as $field) {
            if ($item[$field] !== '') {
                $sku_data[$field] = $this->number($item[$field]);
            }
        }
        if ($item['stock'] !== '') {
            $sku_data['count'] = $this->number($item['stock']);
        }
        $data = [
            'name' => $item['name'],
            'currency' => $currency,
            'status' => 1,
            'type_id' => $type ? (int)$type['id'] : 0,
            'skus' => [-1 => $sku_data],
        ];
        $product = new shopProduct();
        if (!$product->save($data)) {
            throw new waException('Webasyst не сохранил новый товар.');
        }
        $product_id = (int)$product->getId();
        $skus = $product->skus;
        if (!$product_id || !$skus) {
            throw new waException('Не удалось определить SKU созданного товара.');
        }
        $sku = reset($skus);
        return [$product_id, (int)$sku['id']];
    }

    private function updateProduct($product_id, $sku_id, array $item)
    {
        $product = new shopProduct((int)$product_id);
        $data = ['skus' => [(int)$sku_id => []]];
        if ($item['name'] !== '') {
            $data['name'] = $item['name'];
        }
        foreach (['price', 'purchase_price', 'compare_price'] as $field) {
            if ($item[$field] !== '') {
                $data['skus'][(int)$sku_id][$field] = $this->number($item[$field]);
            }
        }
        if ($item['stock'] !== '') {
            $data['skus'][(int)$sku_id]['count'] = $this->number($item['stock']);
        }
        if (!$data['skus'][(int)$sku_id]) {
            unset($data['skus']);
        }
        if (count($data) > 0) {
            $product->save($data);
        }
    }

    private function saveLink($supplier_id, $product_id, $sku_id, array $item, array $raw)
    {
        $data = [
            'supplier_id' => (int)$supplier_id,
            'product_id' => (int)$product_id,
            'sku_id' => (int)$sku_id,
            'supplier_sku' => $item['supplier_sku'],
            'external_id' => $item['external_id'],
            'purchase_price' => $item['purchase_price'] === '' ? null : $this->number($item['purchase_price']),
            'stock' => $item['stock'] === '' ? null : $this->number($item['stock']),
            'raw_json' => json_encode($raw, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES),
            'updated_at' => date('Y-m-d H:i:s'),
        ];
        $existing = $this->map_model->getByField([
            'supplier_id' => (int)$supplier_id,
            'supplier_sku' => $item['supplier_sku'],
        ]);
        if ($existing) {
            $this->map_model->updateById($existing['id'], $data);
        } else {
            $this->map_model->insert($data);
        }
    }

    private function number($value)
    {
        $value = str_replace(["\xc2\xa0", ' ', ','], ['', '', '.'], trim((string)$value));
        if (!is_numeric($value)) {
            throw new waException('Некорректное числовое значение: '.$value);
        }
        return (float)$value;

            $this->getResponse()->setStatus(405);

        $meta = (new shopMegasuppliersMetaModel())->getById('api_key');
        $expected = $meta ? (string)$meta['value'] : '';
        $provided = trim((string)waRequest::server('HTTP_X_MEGASUPPLIERS_KEY', ''));
        if ($expected === '' || $provided === '' || !hash_equals($expected, $provided)) {
            $this->errors = ['unauthorized'];
            $this->getResponse()->setStatus(401);
            return;
        }
        $body = file_get_contents('php://input');
        $payload = json_decode($body, true);
        if (!is_array($payload)) {
            $this->errors = ['invalid_json'];
            $this->getResponse()->setStatus(400);
            return;
        }
        $supplier_code = strtoupper(trim((string)ifset($payload['supplier_code'])));
        if ($supplier_code === '') {
            $this->errors = ['SUPPLIER_REQUIRED'];
            $this->getResponse()->setStatus(422);
            return;
        }
        $supplier = (new shopMegasuppliersSupplierModel())->getByField('code', $supplier_code);
        if (!$supplier || empty($supplier['active'])) {
            $this->errors = ['SUPPLIER_NOT_FOUND'];
            $this->getResponse()->setStatus(422);
            return;
        }
        $items = ifset($payload['items'], []);
        if (!is_array($items) || !$items) {
            $this->errors = ['ITEMS_REQUIRED'];
            $this->getResponse()->setStatus(422);
            return;
        }
        try {
            $service = new shopMegasuppliersImportService();
            $stats = $service->importRows($supplier['id'], $items, [
                'source' => 'api',
                'filename' => '',
                'create_missing' => !empty($payload['create_missing']),
                'update_catalog' => !array_key_exists('update_catalog', $payload) || !empty($payload['update_catalog']),
                'field_map' => ifset($payload['field_map'], []),
            ]);
            $this->response = [
                'status' => 'ok',
                'supplier' => ['id' => (int)$supplier['id'], 'code' => $supplier['code'], 'name' => $supplier['name']],
                'import_id' => (int)$stats['import_id'],

                'linked' => $stats['linked'],
                'skipped' => $stats['skipped'],
                'errors' => $stats['errors'],
            ];

        }
    }
