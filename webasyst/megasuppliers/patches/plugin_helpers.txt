    private function getProductSupplierId($product_id)
    {
        $product_id = (int)$product_id;
        if (!$product_id) {
            return 0;
        }
        $row = (new shopMegasuppliersProductModel())
            ->select('supplier_id')
            ->where('product_id = i:product_id', array('product_id' => $product_id))
            ->order('updated_at DESC, id DESC')
            ->limit(1)
            ->fetchAssoc();
        return $row ? (int)$row['supplier_id'] : 0;
    }

    private function supplierSelectOptions($selected_id)
    {
        $rows = (new shopMegasuppliersSupplierModel())
            ->select('id,name')
            ->where('active=1')
            ->order('name')
            ->fetchAll();
        $html = '<option value="">— не выбран —</option>';
        foreach ($rows as $row) {
            $selected = ((int)$row['id'] === (int)$selected_id) ? ' selected' : '';
            $html .= '<option value="'.(int)$row['id'].'"'.$selected.'>'.htmlspecialchars($row['name'], ENT_QUOTES, 'UTF-8').'</option>';
        }
        return $html;
    }
