    public function productsCollectionFilter($collection)
    {
        if (wa()->getEnv() !== 'backend') {
            return;
        }
        $supplier_id = waRequest::get('megasupplier', 0, waRequest::TYPE_INT);
        if (!$supplier_id) {
            return;
        }
        $supplier = (new shopMegasuppliersSupplierModel())->getById($supplier_id);
        if (!$supplier || empty($supplier['active'])) {
            $collection->addWhere('0');
            return;
        }
        $collection->addWhere(
            'EXISTS (SELECT 1 FROM shop_megasuppliers_product ms '
            .'WHERE ms.product_id=p.id AND ms.supplier_id='.(int)$supplier_id.')'
        );
    }
