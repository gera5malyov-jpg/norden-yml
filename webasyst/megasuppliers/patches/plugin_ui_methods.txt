    private function supplierAssignUrl()
    {
        return wa('shop')->getAppUrl(null, true).'?plugin=megasuppliers&module=backend&action=assignSupplier';
    }

    public function backendProductEdit($product)
    {
        $product_id = (int)$product->id;
        if (!$product_id) {
            return array();
        }
        $selected = $this->getProductSupplierId($product_id);
        $options = $this->supplierSelectOptions($selected);
        $url = htmlspecialchars($this->supplierAssignUrl(), ENT_QUOTES, 'UTF-8');

        $html = '<div class="field s-megasuppliers-legacy-field">'
            .'<div class="name">Поставщик</div>'
            .'<div class="value"><select class="s-megasuppliers-select" data-product-id="'.$product_id.'" data-url="'.$url.'">'
            .$options.'</select> <span class="hint s-megasuppliers-status"></span></div>'
            .'</div>'
            .'<script>(function($){'
            .'var f=$(".s-megasuppliers-legacy-field").last();'
            .'var target=null;'
            .'$(".fields .field .name").each(function(){if($.trim($(this).text()).toUpperCase().indexOf("YML ID")>=0){target=$(this).closest(".field");return false;}});'
            .'if(target&&target.length){f.insertBefore(target);}'
            .'f.find(".s-megasuppliers-select").off("change.ms").on("change.ms",function(){'
            .'var el=$(this),st=f.find(".s-megasuppliers-status");st.text("сохраняю…");'
            .'$.post(el.data("url"),{product_id:el.data("product-id"),supplier_id:el.val(),_csrf:$("input[name=_csrf]").first().val()||""},function(r){'
            .'if(r&&r.status==="ok"){st.text("сохранено");setTimeout(function(){st.text("");},1500);}else{st.text("ошибка");}'
            .'},"json").fail(function(){st.text("ошибка");});'
            .'});'
            .'})(jQuery);</script>';

        return array('basics' => $html);
    }

    public function backendProdContent($params)
    {
        if (empty($params['product']) || ifset($params, 'content_id', '') !== 'general') {
            return array();
        }
        $product = $params['product'];
        $product_id = (int)$product->id;
        if (!$product_id) {
            return array();
        }
        $selected = $this->getProductSupplierId($product_id);
        $options = $this->supplierSelectOptions($selected);
        $url = htmlspecialchars($this->supplierAssignUrl(), ENT_QUOTES, 'UTF-8');

        $html = '<div class="field custom-mt-24 s-megasuppliers-new-field">'
            .'<div class="name"><div class="wa-label for-input"><div>Поставщик</div></div></div>'
            .'<div class="value"><select class="s-megasuppliers-select small" data-product-id="'.$product_id.'" data-url="'.$url.'">'
            .$options.'</select> <span class="hint s-megasuppliers-status"></span></div>'
            .'</div>'
            .'<script>(function($){'
            .'var f=$(".s-megasuppliers-new-field").last();'
            .'var sets=$(".s-product-sets-section").first().closest(".field");'
            .'if(sets.length){f.insertAfter(sets);}'
            .'f.find(".s-megasuppliers-select").off("change.ms").on("change.ms",function(){'
            .'var el=$(this),st=f.find(".s-megasuppliers-status");st.text("сохраняю…");'
            .'$.post(el.data("url"),{product_id:el.data("product-id"),supplier_id:el.val(),_csrf:$("input[name=_csrf]").first().val()||""},function(r){'
            .'if(r&&r.status==="ok"){st.text("сохранено");setTimeout(function(){st.text("");},1500);}else{st.text("ошибка");}'
            .'},"json").fail(function(){st.text("ошибка");});'
            .'});'
            .'})(jQuery);</script>';

        return array('form_bottom' => $html);
    }
