    public function backendProducts($params = array())
    {
        $rows = (new shopMegasuppliersSupplierModel())
            ->select('id,name,code')
            ->where('active=1')
            ->order('name')
            ->fetchAll();
        $base_url = wa('shop')->getAppUrl(null, true).'products/';
        $items = '';
        $options = '<option value="">— поставщик —</option>';
        foreach ($rows as $row) {
            $id = (int)$row['id'];
            $name = htmlspecialchars($row['name'], ENT_QUOTES, 'UTF-8');
            $url = htmlspecialchars($base_url.'?megasupplier='.$id, ENT_QUOTES, 'UTF-8');
            $items .= '<li><a href="'.$url.'"><i class="icon16 folders"></i>'.$name.'</a></li>';
            $options .= '<option value="'.$id.'">'.$name.'</option>';
        }
        $run = htmlspecialchars(wa('shop')->getAppUrl(null, true).'?plugin=megasuppliers&action=importRun', ENT_QUOTES, 'UTF-8');
        $manage = htmlspecialchars(wa('shop')->getAppUrl(null, true).'?plugin=megasuppliers', ENT_QUOTES, 'UTF-8');
        $html = '<div class="block" id="s-megasuppliers-sidebar">'
            .'<span class="count"><a href="'.$manage.'" title="Управление поставщиками"><i class="icon16 settings"></i></a></span>'
            .'<h5 class="heading s-ms-sidebar-toggle" style="cursor:pointer"><i class="icon16 rarr s-ms-sidebar-arrow"></i>Поставщики</h5>'
            .'<div class="s-ms-sidebar-body" style="display:none">'
            .'<ul class="menu-v with-icons">'.$items.'</ul>'
            .'<div class="s-megasuppliers-import-panel" style="margin-top:10px">'
            .'<select class="s-ms-supplier" style="max-width:160px">'.$options.'</select> '
            .'<button type="button" class="button light-gray s-ms-dry-run">Проверить прайс</button>'
            .'<div class="hint s-ms-run-status"></div></div></div></div>'
            .'<script>(function(){'
            .'var root=document.getElementById("s-megasuppliers-sidebar");if(!root)return;'
            .'var toggle=root.querySelector(".s-ms-sidebar-toggle"),body=root.querySelector(".s-ms-sidebar-body"),arrow=root.querySelector(".s-ms-sidebar-arrow");'
            .'if(toggle&&body){toggle.addEventListener("click",function(e){if(e.target&&e.target.closest&&e.target.closest("a"))return;var open=body.style.display!=="none";body.style.display=open?"none":"block";if(arrow){arrow.classList.toggle("rarr",open);arrow.classList.toggle("darr",!open);}});}'
            .'var btn=root.querySelector(".s-ms-dry-run");if(btn){btn.addEventListener("click",function(){'
            .'var id=root.querySelector(".s-ms-supplier").value,st=root.querySelector(".s-ms-run-status");if(!id){st.textContent="выберите поставщика";return;}st.textContent="запуск…";'
            .'var csrf=document.querySelector("input[name=_csrf]");var bodyData=new URLSearchParams();bodyData.set("supplier_id",id);bodyData.set("mode","dry-run");bodyData.set("_csrf",csrf?csrf.value:"");'
            .'fetch("'.$run.'",{method:"POST",credentials:"same-origin",headers:{"Content-Type":"application/x-www-form-urlencoded; charset=UTF-8"},body:bodyData.toString()}).then(function(r){return r.json();}).then(function(r){st.textContent=r&&r.status==="ok"?"проверка запущена":"ошибка запуска";}).catch(function(){st.textContent="ошибка запуска";});'
            .'});}'
            .'})();</script>';
        if (method_exists($this, 'thumbnailBootstrap')) {
            $html .= $this->thumbnailBootstrap();
        }
        return array('sidebar_section' => $html);
    }
